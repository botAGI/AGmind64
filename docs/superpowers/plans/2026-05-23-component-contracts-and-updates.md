```python
"""Pydantic models for component-level version contracts."""

from __future__ import annotations

import re
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, field_validator

ComponentKind = Literal[
    "core",
    "backend",
    "app_stack",
    "stateful_service",
    "edge",
    "ops",
    "deploy_target",
    "model_family",
]
UpdatePolicy = Literal[
    "free",
    "patch-auto",
    "minor-review",
    "major-hold",
    "grouped",
    "pinned-by-parent",
    "volatile",
]

_ID_RE = re.compile(r"^[a-z][a-z0-9-]{1,63}$")
_CAP_RE = re.compile(r"^[a-z][a-z0-9_]{1,63}$")


class ComponentRequirement(BaseModel):
    """One required service capability or external dependency."""

    model_config = ConfigDict(frozen=True, extra="forbid")

    capability: str | None = None
    package: str | None = None
    min_version: str | None = None
    recommended_version: str | None = None
    reason: str = ""

    @field_validator("capability")
    @classmethod
    def _check_capability(cls, value: str | None) -> str | None:
        if value is not None and not _CAP_RE.match(value):
            raise ValueError("capability must be snake_case")
        return value


class ComponentCore(BaseModel):
    """Version policy for the upstream tool or component family."""

    model_config = ConfigDict(frozen=True, extra="forbid")

    upstream: str = ""
    min_version: str | None = None
    recommended_version: str
    current_pin: str | None = None
    update_policy: UpdatePolicy
    hold_reason: str = ""
    source: Literal["registry", "github_release", "pypi", "manual", "local"] = "registry"


class ComponentRuntime(BaseModel):
    """Concrete runtime artifacts owned by this component."""

    model_config = ConfigDict(frozen=True, extra="forbid")

    service_descriptors: tuple[str, ...] = ()
    dockerfiles: tuple[str, ...] = ()
    python_packages: tuple[str, ...] = ()
    ansible_collections: tuple[str, ...] = ()
    model_catalogs: tuple[str, ...] = ()


class ComponentVerification(BaseModel):
    """Commands required before accepting an update."""

    model_config = ConfigDict(frozen=True, extra="forbid")

    commands: tuple[str, ...] = ()
    ci_jobs: tuple[str, ...] = ()
    smoke_profiles: tuple[str, ...] = ()


class ComponentContract(BaseModel):
    """Single source of truth for one component's version/update contract."""

    model_config = ConfigDict(frozen=True, extra="forbid")

    id: str
    kind: ComponentKind
    owner: str = "agmind-core"
    core: ComponentCore
    runtime: ComponentRuntime = Field(default_factory=ComponentRuntime)
    provides: tuple[str, ...] = ()
    requires: tuple[ComponentRequirement, ...] = ()
    optional_providers: dict[str, tuple[str, ...]] = Field(default_factory=dict)
    conflicts: dict[str, tuple[str, ...]] = Field(default_factory=dict)
    verification: ComponentVerification = Field(default_factory=ComponentVerification)

    @field_validator("id")
    @classmethod
    def _check_id(cls, value: str) -> str:
        if not _ID_RE.match(value):
            raise ValueError("id must match ^[a-z][a-z0-9-]{1,63}$")
        return value

    @field_validator("provides")
    @classmethod
    def _check_provides(cls, values: tuple[str, ...]) -> tuple[str, ...]:
        for value in values:
            if not _CAP_RE.match(value):
                raise ValueError(f"provides entry {value!r} must be snake_case")
        return values
```

```python
"""Load component contracts from templates/components/*.yaml."""

from __future__ import annotations

from pathlib import Path

import yaml

from agmind.components.contracts import ComponentContract

REPO_ROOT = Path(__file__).resolve().parents[2]
DEFAULT_COMPONENTS_DIR = REPO_ROOT / "templates" / "components"


def load_component_contracts(
    components_dir: Path = DEFAULT_COMPONENTS_DIR,
) -> dict[str, ComponentContract]:
    """Load all component contracts, keyed by component id."""
    if not components_dir.exists():
        return {}

    out: dict[str, ComponentContract] = {}
    for path in sorted(components_dir.glob("*.yaml")):
        data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
        contract = ComponentContract.model_validate(data)
        if path.stem != contract.id:
            raise ValueError(f"{path.name}: id {contract.id!r} must match filename stem")
        if contract.id in out:
            raise ValueError(f"duplicate component id {contract.id!r}")
        out[contract.id] = contract
    return out
```

"""Component and deploy-level consistency checks."""

from __future__ import annotations

from dataclasses import dataclass

from agmind.schemas import ServiceDescriptor


@dataclass(frozen=True)
class DeployIssue:
    """One deploy-level compatibility issue."""

    severity: str
    kind: str
    services: tuple[str, ...]
    detail: str
    message: str


@dataclass(frozen=True)
class DeployReport:
    """Deploy-level compatibility result."""

    issues: tuple[DeployIssue, ...]

    @property
    def has_errors(self) -> bool:
        return any(issue.severity == "error" for issue in self.issues)


def _host_port(port_spec: str) -> str | None:
    parts = port_spec.split(":")
    if len(parts) == 2:
        return parts[0]
    if len(parts) == 3:
        return parts[1]
    return None


def check_deploy_conflicts(
    selected: dict[str, ServiceDescriptor],
) -> DeployReport:
    """Check conflicts that only exist once services are deployed together."""
    port_owners: dict[str, list[str]] = {}
    for name, descriptor in selected.items():
        for port_spec in descriptor.ports:
            port = _host_port(port_spec)
            if port is None:
                continue
            port_owners.setdefault(port, []).append(name)

    issues: list[DeployIssue] = []
    for port, owners in sorted(port_owners.items()):
        unique = tuple(sorted(set(owners)))
        if len(unique) <= 1:
            continue
        issues.append(
            DeployIssue(
                severity="error",
                kind="host_port_conflict",
                services=unique,
                detail=port,
                message=f"Host port {port} is published by: {', '.join(unique)}",
            )
        )

    return DeployReport(issues=tuple(issues))
```

#!/usr/bin/env python3
"""Validate component contracts and deploy-level conflicts."""

from __future__ import annotations

import sys
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(REPO_ROOT))

from agmind.components.checks import check_deploy_conflicts
from agmind.components.registry import load_component_contracts
from agmind.services.renderer import load_descriptors, select_services


def main() -> int:
    contracts = load_component_contracts()
    descriptors = load_descriptors()

    errors: list[str] = []
    for contract in contracts.values():
        for service_name in contract.runtime.service_descriptors:
            if service_name not in descriptors:
                errors.append(f"{contract.id}: missing service descriptor {service_name}")

    profile_sets = [
        ["core"],
        ["core", "rag"],
        ["core", "observability"],
        ["core", "ragflow"],
    ]
    for profiles in profile_sets:
        selected = select_services(descriptors, profiles=profiles)
        report = check_deploy_conflicts(selected)
        if report.has_errors:
            for issue in report.issues:
                errors.append(f"{','.join(profiles)}: {issue.message}")

    if errors:
        print("Component contract validation failed:", file=sys.stderr)
        for error in errors:
            print(f"- {error}", file=sys.stderr)
        return 1

    print(f"component contracts OK: {len(contracts)} contracts, {len(descriptors)} services")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

Create `docs/adr/0013-component-contracts-and-safe-updates.md`:

```markdown
# ADR-0013: Component Contracts and Safe Updates

- **Status:** accepted
- **Date:** 2026-05-23
- **Driver:** AGmind needs explicit ownership and version policies per tool before adding Proxmox, ComfyUI, Kubernetes, and enterprise service packs.

## Decision

AGmind separates container runtime descriptors from component version contracts.
`templates/services/*.yaml` describes one service. `templates/components/*.yaml`
describes an upstream tool, app stack, backend, model family, deploy target, or
ops stack.

Component contracts own:

- recommended versions and minimum supported versions;
- update policy: `free`, `patch-auto`, `minor-review`, `major-hold`, `grouped`,
  `pinned-by-parent`, or `volatile`;
- grouped runtime artifacts such as all Dify containers;
- required capabilities and optional providers;
- verification commands required before accepting an update.

## Consequences

- `version_check.py` can produce an issue-66-style report that is actionable by component.
- `agmind upgrade` can plan grouped updates and rollback all changed descriptors together.
- service-level compatibility remains soft; deploy-level conflicts such as host ports are checked separately.
- future homelab/enterprise tools must add a component contract before becoming a first-class profile.
```
